import UI/MindstoneButton
import Cosmetics/WitchBroom
import Hats/WitchHat

var gear = import lib/Gear
var bosses = import lib/Bosses
var harv = import lib/Harvest
var travel = import lib/Travel
var combat = import lib/Combat

// Prints the target's id, state and time
// top-left, for calibrating lib/Bosses.
var debug = true

// Dysangelos's talk, auto-skipped. He is
// the only one who talks in the Rocky
// Plateau.
?loc = rocky
  disable npcDialog
:
  enable npcDialog

?item.left.state=3
  equipL st
  equipL @item.left@
?item.right.state=3
  equipR sh *0*
  equipR @item.right@

// A boss is safe only while asleep or
// revealing itself (state 0): state 1 is
// already the idle right before the
// first attack.
var safe
safe = !foe | (foe.distance>20 & foe ! boss)
^| (foe = boss & foe.state=0 & hp = maxhp)
// Never once a boss fight is on: a far
// summoned minion would have the stones
// walk the player into the boss.
?bosses.fightOn & foe
  safe = false

// Highest priority first: each hand goes
// to the first module that asks for it.
gear.Reset()
bosses.Dodge()
gear.Hold()
?safe
  harv.Run()
  travel.Run()
:
  combat.Run()
gear.Apply()

?debug
  >`0,1,L @item.left@ / R @item.right@
  >`0,5,cast: @gear.last@
?debug & foe
  >`0,2,@foe.id@ s@foe.state@ t@foe.time@
  ?foe.debuffs.count > 0
    >`0,3,@foe.debuffs.string@
  ?foe = boss
    >`0,4,dodge: @bosses.info@
  // Whether the Towering Shield is found.
  ?foe = dysangelos_elementalist
    var towers = item.GetCount(combat.tower)
    >`0,6,tower: @towers@ armor: @armor@

//?foe=boss & foe.hp<foe.maxhp
