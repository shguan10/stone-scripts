import UI/MindstoneButton
import Cosmetics/WitchBroom
import Hats/WitchHat

var gear = import lib/Gear
var bosses = import lib/Bosses
var harv = import lib/Harvest
var travel = import lib/Travel
var combat = import lib/Combat

// Prints the target's id, state and time
// top-left, for calibrating lib/Bosses.
var debug = true

?item.left.state=3
  equipL st
  equipL @item.left@
?item.right.state=3
  equipR sh *0*
  equipR @item.right@

var safe
safe = !foe | (foe.distance>20 & foe ! boss)
^| (foe = boss & foe.state<=1 & hp = maxhp)

// Highest priority first: each hand goes
// to the first module that asks for it.
gear.Reset()
bosses.Dodge()
gear.Continue()
?safe
  harv.Run()
  travel.Run()
:
  combat.Run()
gear.Apply()

?debug & foe
  >`0,1,@foe.id@ s@foe.state@ t@foe.time@
  >`0,2,@foe.debuffs.string@
