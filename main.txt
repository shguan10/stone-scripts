import UI/MindstoneButton
import Cosmetics/WitchBroom
import Hats/WitchHat

var gear = import lib/Gear
var bosses = import lib/Bosses
var harv = import lib/Harvest
var travel = import lib/Travel
var combat = import lib/Combat
var element = import lib/Element

// Prints the target's id, state and time
// top-left, for calibrating lib/Bosses.
var debug = true

// When this location began, for the
// start-of-location tower.
var locStart = -999
var startFrames = 2
var dysEls = ["fire", "ice", "aether", "vigor", "poison"]

// Dysangelos's talk, auto-skipped. He is
// the only one who talks in the Rocky
// Plateau.
?loc = rocky
  disable npcDialog
:
  enable npcDialog

?item.left.state=3
  equipL st
  equipL @item.left@
?item.right.state=3
  equipR sh *0*
  equipR @item.right@

// A boss is safe only while asleep or
// revealing itself (state 0): state 1 is
// already the idle right before the
// first attack.
var safe
safe = !foe | (foe.distance>20 & foe ! boss)
^| (foe = boss & foe.state=0 & hp = maxhp)
// Never once a boss fight is on: a far
// summoned minion would have the stones
// walk the player into the boss.
?bosses.fightOn & foe
  safe = false
// Fighting stays on for safeLag frames
// after the last unsafe one: in a crowd
// the target flips between a near foe
// and a far or dormant one, and the
// right hand flickered between the
// Triskelion and the wand (recorded).
var safeLag = 30
var unsafeAt = -999
?loc.begin | loc.loop
  unsafeAt = -999
?!safe
  unsafeAt = totaltime
?totaltime - unsafeAt < safeLag
  safe = false
// Dysangelos before his first attack, and
// between phases, walks safely (Shane):
// the intro holds him in state 100 for
// seconds (recorded).
?bosses.DysIdle()
  safe = true

// Highest priority first: each hand goes
// to the first module that asks for it.
gear.Reset()
// The Towering Shield in hand on a
// location's first frames starts it at
// full armor (Shane).
?loc.begin | loc.loop
  locStart = totaltime
?totaltime - locStart <= startFrames
  gear.R(combat.tower)
gear.Probe()
bosses.Dodge()
gear.Hold()
// Between boss phases (the Shroom
// splitting into Morel) nothing is
// targeted: the counter shield, not the
// walking stones (Shane).
?safe & bosses.ShroomGap()
  gear.R(combat.Shield(element.Best()))
?safe
  harv.Run()
  travel.Run()
:
  combat.Run()
gear.Apply()

?debug
  // Each hand on its own row: one line for
  // both cut the right hand's id off.
  >`0,1,L @item.left@
  >`0,10,R @item.right@
  >`0,5,cast: @gear.last@
  // The star levels Gear's probe found.
  >`0,11,stars v3: @gear.found@
  // How many items each shield search
  // matches: names are cut off on screen.
  var nDash
  var nBash
  var nPlain
  var nTower
  nDash = item.GetCount("dashing shield")
  nBash = item.GetCount("bashing shield")
  nPlain = item.GetCount("shield -dashing -bashing -tower")
  nTower = item.GetCount("tower")
  >`0,9,dash @nDash@ bash @nBash@ plain @nPlain@ tower @nTower@
?debug & foe
  >`0,2,@foe.id@ s@foe.state@ t@foe.time@ d@foe.distance@ n@foe.count@
  ?foe.debuffs.count > 0
    >`0,3,@foe.debuffs.string@
  ?foe = boss
    >`0,4,dodge: @bosses.info@
    >`0,7,buffs: @foe.buffs.string@ cap: @combat.armorCap@
  // Whether the Towering Shield is found.
  ?foe = dysangelos
    // A var's initializer runs only once
    // here, so assign on its own line.
    var towers
    towers = item.GetCount(combat.tower)
    >`0,6,tower: @towers@ armor: @armor@
  // Dysangelos P3's absorbed elements, by
  // name, with frames left: the HUD shows
  // them with the debuffs' symbols.
  ?foe = dysangelos_perfected
    var res
    var left
    res = ""
    for el : dysEls
      left = foe.buffs.GetTime("adaptive_defense_" + el)
      ?left > 0
        res = res + el + " " + left + "  "
    >`0,8,resists: @res@

//?foe=boss & foe.hp<foe.maxhp
  //activate potion
