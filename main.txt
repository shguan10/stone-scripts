func Cast(wep)
  ?item.GetCooldown(wep) <=0
  ^&item.CanActivate()
    equip @wep@
    activate R
    >@wep@
    return true
  :?item.right.id=@wep@ & item.right.state=2
    >@wep@,2
    return true
  return false
