// Bosses: backflip out of the attacks
// that can be avoided, by equipping the
// Mind Stone just before they land.
//
// Frame numbers are community values
// (livercat/scribble, alexbarker234,
// Fuzznimp, the Miraheze wiki) for the
// unchilled boss, and are used only for
// the first attack of each kind. After
// that the dodge comes from how long that
// state actually lasted last time (Track),
// corrected for the change in chill
// stacks (~7 frames each). info says what
// the last dodge did; main prints it in
// debug. Attacks not listed (Hrimnir's
// blows, Dysangelos P2's arms) cannot be
// dodged; the counter shield takes them.

var gear = import lib/Gear

// Dodge this many frames before the
// measured end of the state.
var lead = 4

var info = ""
var win = ""

// Measured state lengths, keyed
// "<foe id>:<state>", with the chill
// stacks during each.
var lenKeys = []
var lens = []
var lenChill = []
var curKey = ""
var curMax = 0
var curChill = 0

func Stacks()
  return foe.debuffs.GetCount("debuff_chill")

func Track()
  ?loc.begin | loc.loop
    lenKeys.Clear()
    lens.Clear()
    lenChill.Clear()
  var sk = foe.id + ":" + foe.state
  ?string.Equals(sk, curKey)
    curMax = foe.time
    ?Stacks() > curChill
      curChill = Stacks()
    return
  // The state just ended: keep its length.
  ?curMax > 0
    var i = lenKeys.IndexOf(curKey)
    ?i < 0
      lenKeys.Add(curKey)
      lens.Add(curMax + 1)
      lenChill.Add(curChill)
    :
      lens.Emplace(i, curMax + 1)
      lenChill.Emplace(i, curChill)
  curKey = sk
  curMax = 0
  curChill = Stacks()

// Inside a 6-frame window starting at
// frame t of the given state, or lead
// frames before its measured end.
func At(st, t)
  ?foe.state ! st
    return false
  var from = t + 7 * Stacks()
  var i = lenKeys.IndexOf(foe.id + ":" + st)
  ?i >= 0
    from = lens[i] - lead + 7 * (Stacks() - lenChill[i])
  ?foe.time = 0
    win = "s" + st + " dodge@" + from
    info = win
  return foe.time >= from & foe.time <= from + 6

func Mind()
  ?item.GetCooldown("mind") > 0
    // Already used on this attack?
    ?info ! "DODGED"
      info = win + " mind on cooldown"
    return false
  // A backflip into the screen edge
  // goes nowhere.
  ?screen.FromWorldX(pos.x) <= 5
    info = win + " at screen edge"
    return false
  ?gear.L("mind")
    info = win + " DODGED t" + foe.time
    return true
  info = win + " hand taken"
  return false

// Returns true if this frame dodges.
// A boss has been seen in this location.
// Stays set while a summoned minion is
// the target, so Travel does not dash at
// it into the boss.
var fightOn = false
// Palas (either stage) has been seen: the
// Blade summons him, so it never fires
// in his fight, minions included.
var palasSeen = false

func Dodge()
  ?loc.begin | loc.loop
    fightOn = false
    palasSeen = false
  ?foe ! boss
    return false
  fightOn = true
  ?foe = skeleton_boss
    palasSeen = true
  Track()
  var fid = foe.id
  // Dysangelos (Rocky)
  ?fid = dysangelos_bearer
    // Eye laser; quicker from 11 stars.
    ?loc.stars >= 11 & At(32, 22)
      return Mind()
    ?loc.stars < 11 & At(32, 28)
      return Mind()
    return false
  ?fid = dysangelos_perfected
    ?At(115, 60)
      return Mind()
    // Stun slam: the 32 that does 1 damage.
    ?foe.damage <= 1 & At(32, 58)
      return Mind()
    return false
  // Xyloalgia (Deadwood). Poena has no
  // dodge here; Debuff stops feeding her
  // mirror.
  ?fid = tree_boss
    ?At(32, 31)
      return Mind()
    return false
  // Bolesh (Caves): the bite.
  ?fid = spider_boss
    ?At(142, 18)
      return Mind()
    return false
  // Angry Shroom (Mushroom): the punch.
  // Morel first: its id contains the
  // other's.
  ?fid = mushroom_boss_fat
    ?At(32, 26)
      return Mind()
    return false
  ?fid = mushroom_boss
    ?At(32, 46)
      return Mind()
    return false
  // Pallas (Halls): phase 2 first, for
  // the same reason.
  ?fid = skeleton_boss_stage_2
    ?At(32, 17)
      return Mind()
    return false
  // Phase 1's blade slam winds up 125
  // frames at 3 stars, 90 at 4, 70 at 5+.
  ?fid = skeleton_boss
    var slam = 68
    ?loc.stars <= 3
      slam = 123
    :?loc.stars = 4
      slam = 88
    ?At(32, slam)
      return Mind()
    return false
  // Bronze Guardian (Mine): the hammer,
  // incl. the 5-star bomb hammer. Immune
  // to chill, so no drift.
  ?fid = bronze_guardian
    ?At(32, 30)
      return Mind()
    return false
  // Hrimnir (Icy): the snowball, close up.
  ?fid = yeti
    ?foe.distance <= 6 & At(32, 23)
      return Mind()
    return false
  // Nagaraja (Temple): the brick, which
  // wipes all armor.
  ?fid = nagaraja
    ?At(112, 45)
      return Mind()
    return false
  return false
