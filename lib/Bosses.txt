// Bosses: backflip out of the attacks
// that can be avoided, by equipping the
// Mind Stone just before they land.
//
// Frame numbers are community values
// (livercat/scribble, alexbarker234,
// Fuzznimp, the Miraheze wiki) for the
// unchilled boss, and are used only for
// the first attack of each kind. After
// that the dodge comes from how long that
// state actually lasted last time (Track),
// corrected for the change in chill
// stacks (~7 frames each). info says what
// the last dodge did; main prints it in
// debug. Attacks not listed (Hrimnir's
// blows, Dysangelos P2's arms) cannot be
// dodged; the counter shield takes them.

var gear = import lib/Gear

// Dodge this many frames before the
// measured end of the state.
var lead = 4

var info = ""
var win = ""

// Measured state lengths, keyed
// "<foe id>:<state>", with the chill
// stacks during each.
var lenKeys = []
var lens = []
var lenChill = []
var curKey = ""
var curMax = 0
var curChill = 0

func Stacks()
  return foe.debuffs.GetCount("debuff_chill")

// Dysangelos asleep, waking, or between
// phases: abilities would be wasted. Each
// phase counts as begun at his first
// attack (state 2, 32 or 33): the intro
// passes through states no list caught
// (126, 118 in a recording). The states
// below also count as idle afterwards
// (alexbarker234's script).
var dysIdle = [-1, 0, 102, 116, 117, 124, 126, 127]
var dysAttacks = [2, 32, 33]
var dysActive = ""

func DysIdle()
  ?string.IndexOf(foe.id, "dysangelos") < 0
    return false
  ?dysAttacks.Contains(foe.state)
    dysActive = foe.id
  ?!string.Equals(dysActive, foe.id)
    return true
  ?dysIdle.Contains(foe.state)
    return true
  ?foe = dysangelos_bearer
    return foe.state = 100 | foe.state = 101
  return foe = dysangelos_elementalist & foe.state = 107

func Track()
  ?loc.begin | loc.loop
    lenKeys.Clear()
    lens.Clear()
    lenChill.Clear()
  var sk = foe.id + ":" + foe.state
  ?string.Equals(sk, curKey)
    curMax = foe.time
    // The stacks at the state's end, not
    // the most seen: chill decays during a
    // long wind-up, and the max (4, ending
    // at 1) made the next dodge ~21 frames
    // early (recorded, Palas).
    curChill = Stacks()
    return
  // The state just ended: keep its length.
  ?curMax > 0
    var i = lenKeys.IndexOf(curKey)
    ?i < 0
      lenKeys.Add(curKey)
      lens.Add(curMax + 1)
      lenChill.Add(curChill)
    :
      lens.Emplace(i, curMax + 1)
      lenChill.Emplace(i, curChill)
  curKey = sk
  curMax = 0
  curChill = Stacks()

// Inside a 6-frame window starting at
// frame t of the given state, or lead
// frames before its measured end.
func At(st, t)
  ?foe.state ! st
    return false
  var from = t + 7 * Stacks()
  var i = lenKeys.IndexOf(foe.id + ":" + st)
  ?i >= 0
    from = lens[i] - lead + 7 * (Stacks() - lenChill[i])
  ?foe.time = 0
    win = "s" + st + " dodge@" + from
    info = win
  return foe.time >= from & foe.time <= from + 6

// At() without the chill correction, for
// a wind-up chill barely slows: Palas's
// slam ran ~70 frames unchilled and 70-74
// with chill decaying 4 -> 1 (recorded),
// so 7 per stack put dodges up to 30
// frames off. The learned length as is,
// with a longer lead (t60 and t66 both
// avoided the hit landing at ~72).
var flatLead = 8

func AtFlat(st, t)
  ?foe.state ! st
    return false
  var from = t
  var i = lenKeys.IndexOf(foe.id + ":" + st)
  ?i >= 0
    from = lens[i] - flatLead
  ?foe.time = 0
    win = "s" + st + " dodge@" + from + " flat"
    info = win
  return foe.time >= from & foe.time <= from + 6

func Mind()
  ?item.GetCooldown("mind") > 0
    // Already used on this attack?
    ?info ! "DODGED"
      info = win + " mind on cooldown"
    return false
  // A backflip into the screen edge
  // goes nowhere.
  ?screen.FromWorldX(pos.x) <= 5
    info = win + " at screen edge"
    return false
  ?gear.L("mind")
    info = win + " DODGED t" + foe.time
    return true
  info = win + " hand taken"
  return false

// Dysangelos P3's orb (state 115): its
// beam is warned of by a "!" over the
// player (Fuzznimp), else is dodged at a
// fixed frame 70 (livercat, Fuzznimp).
// Not At(): the beam is the orb's, so
// neither the cast's length nor chill
// says when it fires.
func Warned()
  var x = screen.FromWorldX(pos.x)
  var y = screen.FromWorldZ(pos.z - pos.y)
  return string.Equals(draw.GetSymbol(x + 1, y - 8), "!")

func Orb()
  ?foe.time = 0
    win = "s115 orb"
    info = win
  ?foe.time >= 25 & Warned()
    win = "s115 orb !"
    return Mind()
  ?foe.time >= 70 & foe.time <= 76
    win = "s115 orb t70"
    return Mind()
  return false

// Returns true if this frame dodges.
// A boss has been seen in this location.
// Stays set while a summoned minion is
// the target, so Travel does not dash at
// it into the boss.
var fightOn = false
// Palas (either stage) has been seen: the
// Blade summons him, so it never fires
// in his fight, minions included.
var palasSeen = false
// The last boss-tagged foe's id, for the
// gaps between boss phases.
var lastBoss = ""

// Between the Mushroom Forest's phases:
// the Shroom or Morel was the last boss
// and has gone (Ekki is the last phase;
// the minibosses are boss-tagged too).
func ShroomGap()
  ?foe
    return false
  return string.Equals(lastBoss, "mushroom_boss") |
  ^string.Equals(lastBoss, "mushroom_boss_fat")

func Dodge()
  ?loc.begin | loc.loop
    fightOn = false
    palasSeen = false
    lastBoss = ""
    dysActive = ""
  ?foe ! boss
    return false
  fightOn = true
  lastBoss = foe.id
  ?foe = skeleton_boss
    palasSeen = true
  Track()
  var fid = foe.id
  // Dysangelos (Rocky) P1: nothing. His
  // eye laser (every 6th attack, landing
  // at 33 t24) cannot be dodged (Shane),
  // and the arms are not worth the Mind
  // Stone.
  ?fid = dysangelos_bearer
    return false
  // The orb's beam is the only dodge: the
  // stun slam does 1 damage and stuns
  // only up close, and the Mind Stone
  // must be off cooldown for the beam.
  ?fid = dysangelos_perfected
    ?foe.state = 115
      return Orb()
    return false
  // Xyloalgia (Deadwood). Poena has no
  // dodge here; Debuff stops feeding her
  // mirror.
  ?fid = tree_boss
    ?At(32, 31)
      return Mind()
    return false
  // Bolesh (Caves): the bite.
  ?fid = spider_boss
    ?At(142, 18)
      return Mind()
    return false
  // Angry Shroom (Mushroom): the punch.
  // Morel first: its id contains the
  // other's.
  // Morel's 32 lasts 22 frames unchilled
  // (recorded), so the community 26 never
  // came: 22 less the lead.
  ?fid = mushroom_boss_fat
    ?At(32, 18)
      return Mind()
    return false
  ?fid = mushroom_boss
    ?At(32, 46)
      return Mind()
    return false
  // Pallas (Halls): phase 2 first, for
  // the same reason.
  ?fid = skeleton_boss_stage_2
    ?At(32, 17)
      return Mind()
    return false
  // Phase 1's blade slam winds up 125
  // frames at 3 stars, 90 at 4, 70 at 5+
  // (recorded ~70 unchilled at 5+, the hit
  // at 33 t2): 66 is that less the lead.
  ?fid = skeleton_boss
    var slam = 66
    ?loc.stars <= 3
      slam = 123
    :?loc.stars = 4
      slam = 88
    ?AtFlat(32, slam)
      return Mind()
    return false
  // Bronze Guardian (Mine): the hammer,
  // incl. the 5-star bomb hammer. Immune
  // to chill, so no drift.
  ?fid = bronze_guardian
    ?At(32, 30)
      return Mind()
    return false
  // Hrimnir (Icy): the snowball, close up.
  ?fid = yeti
    ?foe.distance <= 6 & At(32, 23)
      return Mind()
    return false
  // Nagaraja (Temple): the brick, which
  // wipes all armor.
  ?fid = nagaraja
    ?At(112, 45)
      return Mind()
    return false
  return false
