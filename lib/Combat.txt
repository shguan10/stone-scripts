// Combat: everything once a foe is close.
// Offensive abilities first. Then a wand
// and a shield of the element that
// counters the foe; against a boss a
// debuff wand can take the shield's place.
// Crossbows and melee (see Melee) are
// switched off for now: the calls are
// commented out in Run. The dashing
// shields belong to melee only.

var gear = import lib/Gear
var element = import lib/Element
var debuff = import lib/Debuff
var melee = import lib/Melee
var travel = import lib/Travel
var bosses = import lib/Bosses

var arms = ["skeleton arm", "skeleton_arm"]
var fallbackWands = ["vigor wand", "wand"]
var tower = "towering shield"
var towerArmor = 6

func Abilities()
  ?foe.distance > 20
    return
  // Plague Wand: weaken, if not on yet.
  ?foe = boss & debuff.Needs("debuff_damage")
    gear.Cast("wand_poison", "poison wand", "L")
  // Feeble, on bosses only. Nagaraja
  // cannot be masked. No range limit:
  // wands fight from beyond melee range.
  ?foe = boss & foe ! nagaraja
    gear.Cast("mask", "mask", "R")
  // Skeleton Arm: whenever off cooldown,
  // at any range; whether it can fire is
  // the game's call, and Gear puts the
  // item back if it cannot. Switched off
  // for now.
  //?foe ! immune_to_physical
  //  var arm = gear.First(arms)
  //  ?arm
  //    gear.Cast("skeleton_arm", arm, "B")
  // Never once Palas is seen, his minions
  // included: the ability summons Palas,
  // so the game refuses it in his fight.
  ?!bosses.palasSeen
    gear.Cast("blade", "blade", "B")

// The counter wand does the damage. A boss
// missing a debuff gets the wand that
// applies it in place of the shield, of a
// different element (one item cannot be
// in both hands).
func Ranged()
  var el = element.Best()
  var wand = ""
  var wandEl = ""
  var c = ""
  ?el
    c = el + " wand"
    ?gear.Has(c)
      wand = c
      wandEl = el
  ?!wand
    wand = gear.First(fallbackWands)
    ?string.IndexOf(wand, "vigor") = 0
      wandEl = "vigor"
  ?wand
    gear.L(wand)
  // Dysangelos P2: the Towering Shield
  // until armor is back to towerArmor,
  // then the debuffs again.
  ?foe = dysangelos_elementalist &
  ^armor < towerArmor & gear.Has(tower)
    gear.R(tower)
    return
  ?foe = boss
    c = debuff.Wand(wandEl)
    ?string.Size(c) > 0
      gear.R(c)
      return
  // Plain shield: the dashing ones would
  // dash in on their own.
  var shield = "shield -dashing -bashing"
  ?el
    c = el + " shield -dashing -bashing"
    ?gear.Has(c)
      shield = c
  gear.R(shield)

// Two-handed, so both hands or nothing,
// and no debuffs: there is no shield to
// swap out. Without one, Ranged.
func Crossbow(el)
  var c = "crossbow"
  ?el
    c = el + " crossbow"
  // A chilling one first, if there is one.
  ?gear.Has(c + " di")
    c = c + " di"
  ?gear.Has(c)
    gear.Both(c)
  :
    Ranged()

// Wand range, learned: the furthest a
// boss has been when the wand started an
// attack (state 2). Starts low so the
// first approach does not stall short.
var wandRange = 12

func Run()
  ?item.left = wand & item.left.state = 2 &
  ^foe.distance > wandRange
    wandRange = foe.distance
  Abilities()
  // A boss out of wand range is walked to
  // only until it is first hit. After
  // that the gap is usually a backflip,
  // and walking it with the stones in
  // hand would drop the shield. Never a
  // dash: it lands in melee range.
  ?foe = boss & foe.distance > wandRange &
  ^foe.hp = foe.maxhp
    travel.Walk()
    return
  Ranged()
