// Combat: everything once a foe is close.
// Offensive abilities first. Then a wand
// and a shield of the element that
// counters the foe; against a boss a
// debuff wand can take the shield's place.
// Crossbows and melee (see Melee) are
// switched off for now: the calls are
// commented out in Run. The dashing
// shields belong to melee only.

var gear = import lib/Gear
var element = import lib/Element
var debuff = import lib/Debuff
var melee = import lib/Melee
var travel = import lib/Travel
var bosses = import lib/Bosses

var arms = ["skeleton arm", "skeleton_arm"]
var fallbackWands = ["vigor wand", "wand"]
var tower = "tower"
var lifesteal = "vigor wand dl"

// Fought like a boss: the Rocky Plateau's
// Acronian Scout hits for 8-10 every 3 s
// and killed a player on plain-foe gear
// (recorded). Mask and poison both cut
// its damage.
func Tough()
  return foe = boss | Scout()

// The Scout is boss-tagged (the debug
// dodge row shows for it); matched by id
// string, in case a search filter misses.
func Scout()
  return string.Equals(foe.id, "acronian_scout")

func Abilities()
  ?foe.distance > 20 | bosses.DysIdle()
    return
  // Plague Wand: weaken, if not on yet.
  // The hidden wands are found by id: a
  // plain "poison wand" is not it.
  ?Tough() & debuff.Needs("debuff_damage")
    gear.Cast("wand_poison", "wand_hidden_poison", "L")
  // Explosive Wand: its active is off on
  // every boss (Shane); it is still held
  // as a burn debuff wand. It never touched
  // the floating Dysangelos nor the
  // Mushroom bosses.
  //?Tough() & string.IndexOf(foe.id, "dysangelos") < 0 &
  //^foe ! mushroom_boss
  //  gear.Cast("wand_fire", "wand_hidden_fire", "L")
  // Feeble, on bosses only. Nagaraja
  // cannot be masked. No range limit:
  // wands fight from beyond melee range.
  ?Tough() & foe ! nagaraja
    gear.Cast("mask", "mask", "R")
  // Skeleton Arm: whenever off cooldown,
  // at any range; whether it can fire is
  // the game's call, and Gear puts the
  // item back if it cannot. Switched off
  // for now.
  //?foe ! immune_to_physical
  //  var arm = gear.First(arms)
  //  ?arm
  //    gear.Cast("skeleton_arm", arm, "B")
  // Never once Palas is seen, his minions
  // included: the ability summons Palas,
  // so the game refuses it in his fight.
  ?!bosses.palasSeen
    gear.Cast("blade", "blade", "B")

// The armor cap, read from maxarmor only
// while a shield is in hand (it depends
// on the shield). High until then, so the
// first shield goes on.
//
// Full from capGap below the cap until
// it falls capDrop below: armor drains
// slowly without a shield, and a single
// threshold swapped shield and wand every
// frame or two at 5.9/6.0 (recorded).
var armorCap = 99
var capGap = 1
var capDrop = 2
// Dysangelos P2's attacks knock armor
// past capDrop so often that the wands
// rarely came out (Shane), and with the
// tower (cap 14) armor levels off at
// 11-12 under them (recorded), short of
// capGap.
var dysP2Gap = 4
var dysP2Drop = 8
// Palas (both phases): dual wield from
// palasArmor (Shane), back to the shield
// below palasLow. One point apart, the
// hands swapped every ~1.5 s (recorded).
var palasArmor = 3
var palasLow = 1
var armorUp = false

func ArmorFull()
  ?item.right = shield
    armorCap = maxarmor
  var gap = capGap
  var drop = capDrop
  ?foe = dysangelos_elementalist
    gap = dysP2Gap
    drop = dysP2Drop
  :?foe = skeleton_boss
    gap = armorCap - palasArmor
    drop = armorCap - palasLow
  ?armor >= armorCap - gap
    armorUp = true
  :?armor < armorCap - drop
    armorUp = false
  return armorUp

// The counter wand does the damage. Once
// armor is near its cap, a boss gets a
// debuff wand in place of the shield, of
// a different element (one item cannot be
// in both hands): one applying a missing
// debuff, else any.
func Ranged()
  var el = element.Best()
  // Dysangelos P3 resists the element
  // that hurt him most at each shield-up.
  ?foe = dysangelos_perfected
    el = melee.DysWeakness()
  var wand = ""
  var wandEl = ""
  var c = ""
  ?el
    c = el + " wand"
    ?gear.Has(c)
      wand = c
      wandEl = el
  ?!wand
    wand = gear.First(fallbackWands)
    ?string.IndexOf(wand, "vigor") = 0
      wandEl = "vigor"
  // Dysangelos P1 and P3 and the Scout,
  // the left hand (Shane): at full HP, any missing
  // debuff; below it, the lifesteal vigor
  // wand, swapped only for poison when he
  // is not poisoned. P3 keeps the wand of
  // an element he does not resist when he
  // resists vigor.
  ?foe = dysangelos_bearer | foe = dysangelos_perfected | Scout()
    c = ""
    ?hp >= maxhp
      c = debuff.Wand("")
    :?debuff.Needs("debuff_damage") &
    ^gear.Has("poison wand dp")
      c = "poison wand dp"
    :?foe.buffs.string ! adaptive_defense_vigor &
    ^gear.Has(lifesteal)
      c = lifesteal
    ?string.Size(c) > 0
      wand = c
      wandEl = debuff.El(c)
      ?string.Equals(c, lifesteal)
        wandEl = "vigor"
  ?wand
    gear.L(wand)
  // The shield only builds armor, which
  // stays when it is put away: once near
  // the cap, a debuff wand, in every fight
  // (Shane).
  // Only a boss is debuffed by what it is
  // missing: in a crowd the target changes
  // every few frames, each with its own
  // debuffs, and the wand followed it
  // (recorded). Elsewhere the one in hand
  // stays.
  ?ArmorFull()
    c = ""
    ?Tough()
      c = debuff.Wand(wandEl)
    ?string.Size(c) = 0
      c = debuff.AnyWand(wandEl)
    ?string.Size(c) > 0
      gear.R(c)
      return
  gear.R(Shield(el))

// A counter-element shield first (Shane:
// poison beats the tower in the Mushroom
// Forest; the tower has no element
// versions); else the Towering Shield.
// The plain Stone shield never goes on in
// a fight while a tower is owned. The
// dashing ones would dash in on their own.
func Shield(el)
  var shield = tower
  ?el
    var c = el + " shield -dashing -bashing"
    ?gear.Has(c)
      shield = c
  ?!gear.Has(shield)
    shield = "shield -dashing -bashing"
  return shield

// Two-handed, so both hands or nothing,
// and no debuffs: there is no shield to
// swap out. Without one, Ranged.
func Crossbow(el)
  var c = "crossbow"
  ?el
    c = el + " crossbow"
  // A chilling one first, if there is one.
  ?gear.Has(c + " di")
    c = c + " di"
  ?gear.Has(c)
    gear.Both(c)
  :
    Ranged()

// Wand range, learned: the furthest a
// boss has been when the wand started an
// attack (state 2). Starts low so the
// first approach does not stall short.
var wandRange = 12

func Run()
  ?item.left = wand & item.left.state = 2 &
  ^foe.distance > wandRange
    wandRange = foe.distance
  var dys3 = foe = dysangelos_perfected & !bosses.DysIdle()
  // The Heavy Hammer on armor: Dysangelos
  // P3, and the Mushroom bosses (Shane;
  // Morel had 185, recorded).
  ?dys3 | foe = mushroom_boss
    melee.Heavy()
  Abilities()
  // Morel (the Mushroom Forest's second
  // phase) is rushed and meleed, vigor
  // hammer then vigor sword; Bosses still
  // dodges the big punch (Shane).
  ?foe = mushroom_boss_fat
    melee.Run()
    return
  // A boss out of wand range is walked to
  // only until it is first hit. After
  // that the gap is usually a backflip,
  // and walking it with the stones in
  // hand would drop the shield. Never a
  // dash: it lands in melee range.
  ?foe = boss & foe.distance > wandRange &
  ^foe.hp = foe.maxhp
    travel.Approach()
    return
  ?dys3
    ?melee.Dys3Hammer(ArmorFull())
      return
  Ranged()
