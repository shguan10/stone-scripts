// Travel: moving between fights. Dash
// with the Quarterstaff when it is up;
// otherwise walk with the Ouroboros
// (heal) and the Triskelion (speed); the
// Star Stone takes the Triskelion's hand
// to pull in pickups (Shane). Approach,
// for walking up to a boss, holds a
// shield instead: the Dashing, the
// Bashing while Dashing is on cooldown,
// else the Towering.

var gear = import lib/Gear
var bosses = import lib/Bosses

// Not in a boss fight, minions included:
// the dash ends in the boss's melee
// range, and fights stay ranged.
func Dash()
  ?foe = boss | bosses.fightOn
    return false
  ?foe & foe.distance <= 17
    return false
  return gear.Cast("quarterstaff", "quarterstaff", "B")

func Walk()
  gear.L("ouroboros")
  ?pickup & pickup.distance < 10
    gear.R("star")
  :
    gear.R("triskelion")

func Approach()
  gear.L("ouroboros")
  ?item.GetCooldown("dash") <= 0 & gear.Has("dashing shield")
    gear.R("dashing shield")
  :?item.GetCooldown("bash") <= 0 & gear.Has("bashing shield")
    gear.R("bashing shield")
  :?gear.Has("tower")
    gear.R("tower")
  :
    gear.R("triskelion")

func Run()
  Dash()
  Walk()
