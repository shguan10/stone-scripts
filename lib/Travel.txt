// Travel: moving between fights. Dash
// with the Quarterstaff when it is up;
// otherwise walk with the Ouroboros
// (heal) and the Triskelion (speed); the
// Star Stone takes the Triskelion's hand
// to pull in pickups (Shane). Close, for
// getting to wand range in a fight, does
// the same with the Quarterstaff's dash
// allowed at bosses.

var gear = import lib/Gear
var bosses = import lib/Bosses

// Not in a boss fight, minions included:
// the dash ends in the boss's melee
// range, and fights stay ranged.
func Dash()
  ?foe = boss | bosses.fightOn
    return false
  ?foe & foe.distance <= 17
    return false
  return gear.Cast("quarterstaff", "quarterstaff", "B")

func Walk()
  gear.L("ouroboros")
  ?pickup & pickup.distance < 10
    gear.R("star")
  :
    gear.R("triskelion")

// Closing to wand range in a fight, boss
// fights included: the Quarterstaff's
// mini-dash while the target is beyond
// 17 (Dash's own gate), else walking.
func Close()
  ?foe.distance > 17
    gear.Cast("quarterstaff", "quarterstaff", "B")
  Walk()

func Run()
  Dash()
  Walk()
