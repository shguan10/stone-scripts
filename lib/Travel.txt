// Travel: moving between fights. Dash
// with the Quarterstaff when it is up;
// otherwise walk with the Triskelion
// (speed) and the Ouroboros (heal).
// The Star Stone takes the Ouroboros's
// hand to pull in pickups.

var gear = import lib/Gear

// Not at a boss: the dash ends in its
// melee range, and fights stay ranged.
func Dash()
  ?foe = boss
    return false
  ?foe & foe.distance <= 17
    return false
  return gear.Cast("quarterstaff", "quarterstaff", "B")

func Walk()
  ?pickup & pickup.distance < 10
    gear.L("star")
  gear.L("ouroboros")
  gear.R("triskelion")

func Run()
  Dash()
  Walk()

// To a boss on purpose (Palas, to fire
// the Blade): dash from afar, else walk
// fast.
func Rush()
  ?foe.distance > 17
    ?gear.Cast("quarterstaff", "quarterstaff", "B")
      return
  Walk()
