// Debuff: which debuff the target is
// missing, and the weapon that applies it.
//
// Only affixed weapons apply debuffs:
//   dP poison -> debuff_damage (weaken)
//   dF fire   -> debuff_dot (burn)
//   dI ice    -> debuff_chill (max 6)
// The Cultist Mask's Feeble is cast as
// an ability, in Combat.

var gear = import lib/Gear

// Swords first: bosses are fought in
// melee (see Melee); a wand still works.
var poisonOnes = ["poison sword dp", "poison wand dp"]
var fireOnes = ["fire sword df", "fire wand df"]
var iceOnes = ["ice sword di", "ice wand di"]

// Poena's mirror reflects debuffs back;
// Dysangelos's protection blocks them.
func Blocked()
  ?foe.buffs.string = poena_mirror
    return true
  return foe.buffs.string = buff_protection

func Can(id)
  ?Blocked()
    return false
  var tag = "immune_to_" + id
  return foe ! tag

// Below its target stacks: venom (weaken)
// at 1, chill stacked to 5, burn only
// when missing. Venom did not stack past
// 1 in game, though the wiki says weaken
// stacks.
var venomStacks = 1
var chillStacks = 5

func Needs(id)
  ?!Can(id)
    return false
  var n = foe.debuffs.GetCount(id)
  ?id = "debuff_damage"
    return n < venomStacks
  ?id = "debuff_chill"
    return n < chillStacks
  return n = 0

// Poison first, then burn, then chill.
// "" when nothing is missing or no
// owned weapon applies what is. Skips
// element skipEl: the other hand holds
// it, and one item cannot be in both.
func Weapon(skipEl)
  var w = ""
  ?Needs("debuff_damage") &
  ^!string.Equals(skipEl, "poison")
    w = gear.First(poisonOnes)
  ?!w & Needs("debuff_dot") &
  ^!string.Equals(skipEl, "fire")
    w = gear.First(fireOnes)
  ?!w & Needs("debuff_chill") &
  ^!string.Equals(skipEl, "ice")
    w = gear.First(iceOnes)
  return w

// The debuff each wand applies, by index.
var wands = ["poison wand dp", "fire wand df", "ice wand di"]
var wandEls = ["poison", "fire", "ice"]
var wandIds = ["debuff_damage", "debuff_dot", "debuff_chill"]
var anyOrder = [1, 0, 2]

// The index of the debuff wand in the
// right hand, else -1.
func InHand()
  var i = 0
  for w : wands
    ?item.right = w
      return i
    i++
  return -1

// Wands only, for ranged fights. Skips
// element skipEl, as Weapon does. The one
// in hand stays while its debuff is still
// missing: every swap restarts a wand's
// attack.
func Wand(skipEl)
  var h = InHand()
  ?h >= 0
    ?Needs(wandIds[h]) &
    ^!string.Equals(skipEl, wandEls[h])
      return wands[h]
  ?Needs("debuff_damage") &
  ^!string.Equals(skipEl, "poison") &
  ^gear.Has("poison wand dp")
    return "poison wand dp"
  ?Needs("debuff_dot") &
  ^!string.Equals(skipEl, "fire") &
  ^gear.Has("fire wand df")
    return "fire wand df"
  ?Needs("debuff_chill") &
  ^!string.Equals(skipEl, "ice") &
  ^gear.Has("ice wand di")
    return "ice wand di"
  return ""

// A debuff wand for its damage alone,
// when nothing is missing: the one in
// hand, else fire, poison, ice. Skips
// element skipEl.
func AnyWand(skipEl)
  var h = InHand()
  ?h >= 0
    ?!string.Equals(skipEl, wandEls[h])
      return wands[h]
  for j : anyOrder
    ?!string.Equals(skipEl, wandEls[j]) &
    ^gear.Has(wands[j])
      return wands[j]
  return ""
